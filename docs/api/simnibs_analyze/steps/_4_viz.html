 60    @staticmethod
 61    def _build_plotter(
 62        efield_path: Path,
 63        camera_position: str = "xy",
 64        cmap: str = "hot",
 65        threshold_percentile: float = 0.0,
 66        vmin: Optional[float] = None,
 67        vmax: Optional[float] = None,
 68        mask_path: Optional[Path] = None,
 69        mask_color: str = "cyan",
 70        mask_opacity: float = 0.3,
 71        off_screen: bool = True,
 72        component: Union[str, int] = "magnitude",
 73    ):  # -> pv.Plotter (lazy import, not annotated to avoid import at module level)
 74        """
 75        Build and return a configured PyVista Plotter for e-field visualisation.
 76
 77        Shared by :meth:`_create_3d_view` (offscreen static render) and
 78        :meth:`view_efield_interactive` (interactive window).
 79
 80        Parameters
 81        ----------
 82        efield_path :
 83            Path to the e-field NIfTI file.
 84        camera_position :
 85            PyVista camera position string (``'xy'``, ``'xz'``, ``'yz'``).
 86        cmap :
 87            Colormap name.
 88        threshold_percentile :
 89            Voxels below this percentile of non-zero values are zeroed.
 90        vmin, vmax :
 91            Explicit colour scale limits. If both ``None``, PyVista auto-scales.
 92        mask_path :
 93            Optional binary mask NIfTI to colocalize with the e-field
 94            (e.g. ``cereb_mask.nii.gz`` from SimNIBS surfaces/).  Rendered as
 95            a semi-transparent coloured surface in the same space.
 96        mask_color :
 97            Colour of the mask surface mesh (default ``'cyan'``).
 98        mask_opacity :
 99            Opacity of the mask surface mesh (default 0.3).
100        off_screen :
101            If ``True``, creates an offscreen plotter (static PNG).
102            If ``False``, creates an interactive window plotter.
103        component :
104            How to reduce a 4D vector field (Ex/Ey/Ez) to a scalar volume.
105            ``'magnitude'`` (default) → computes ‖E‖ = √(Ex²+Ey²+Ez²).
106            ``0``, ``1``, ``2`` → extracts a single cartesian component
107            (Ex, Ey, Ez respectively); a diverging colormap (``coolwarm``) is
108            used automatically so that negative values are visible.
109            Ignored for 3D (scalar) inputs.
110        """
111        _COMPONENT_LABELS = {0: "Ex", 1: "Ey", 2: "Ez"}
112
113        efield_img = nib.as_closest_canonical(nib.load(str(efield_path)))
114        raw = np.squeeze(efield_img.get_fdata())
115
116        # ── Cas 1 : fichier scalaire 3D (scalar_magnE, …) ─────────────────
117        if raw.ndim == 3:
118            logger.info(f"3D scalar e-field {raw.shape} — loading as-is")
119            data = raw
120            # cmap par défaut : hot (valeurs ≥ 0)
121
122        # ── Cas 2 : champ vectoriel 4D (scalar_E → Ex/Ey/Ez) ─────────────
123        elif raw.ndim == 4:
124            if component == "magnitude":
125                logger.info(
126                    f"4D vector e-field {raw.shape} — computing ‖E‖ "
127                    f"(tip: use scalar_magnE directly for faster loading)"
128                )
129                data = np.linalg.norm(raw, axis=-1)
130                # cmap par défaut : hot (valeurs ≥ 0)
131            elif isinstance(component, int) and 0 <= component <= 2:
132                label = _COMPONENT_LABELS[component]
133                logger.info(
134                    f"4D vector e-field {raw.shape} — extracting component "
135                    f"{label} (vol[{component}]), signed"
136                )
137                data = raw[..., component]
138                if cmap == "hot":
139                    cmap = "coolwarm"  # colormap divergente pour valeurs signées
140            else:
141                raise ValueError(
142                    f"component must be 'magnitude', 0, 1, or 2 — got {component!r}"
143                )
144        else:
145            raise ValueError(f"Unexpected e-field shape: {raw.shape}")
146
147        try:
148            import pyvista as pv
149        except ImportError as exc:
150            raise ImportError(
151                "pyvista is required for 3D rendering. "
152                "Install it with: pip install pyvista"
153            ) from exc
154
155        # Les voxels à 0 (hors-cerveau, masque fond) → NaN = transparents en volume rendering
156        data = data.astype(float)
157        data[data == 0] = np.nan
158
159        efield_spacing = efield_img.header.get_zooms()[:3]
160        efield_origin = efield_img.affine[:3, 3]
161
162        grid = pv.ImageData()
163        grid.dimensions = np.array(data.shape) + 1
164        grid.spacing = efield_spacing
165        grid.origin = efield_origin
166        grid.cell_data["values"] = data.flatten(order="F")
167
168        plotter = pv.Plotter(off_screen=off_screen)
169
170        # ── Mask surface overlay ──────────────────────────────────────────
171        if mask_path is not None:
172            mask_img = nib.as_closest_canonical(nib.load(str(mask_path)))
173            mask_data = np.squeeze(mask_img.get_fdata()).astype(np.float32)
174            mask_spacing = mask_img.header.get_zooms()[:3]
175            mask_origin = mask_img.affine[:3, 3]
176            mask_grid = pv.ImageData()
177            mask_grid.dimensions = np.array(mask_data.shape) + 1
178            mask_grid.spacing = mask_spacing
179            mask_grid.origin = mask_origin
180            mask_grid.cell_data["mask"] = mask_data.flatten(order="F")
181            mask_surface = mask_grid.threshold(0.5).extract_surface()
182            if mask_surface.n_points > 0:
183                plotter.add_mesh(
184                    mask_surface,
185                    color=mask_color,
186                    opacity=mask_opacity,
187                    smooth_shading=True,
188                )
189
190        # ── Volume e-field ────────────────────────────────────────────────
191        if vmin is not None and vmax is not None:
192            plotter.add_volume(grid, cmap=cmap, clim=[vmin, vmax])
193        else:
194            plotter.add_volume(grid, cmap=cmap)
195
196        plotter.camera_position = camera_position
197        return plotter

234    def view_efield_interactive(
235        self,
236        efield_path: Path,
237        mask_path: Optional[Path] = None,
238        mask_color: str = "cyan",
239        mask_opacity: float = 0.3,
240        threshold_percentile: Optional[float] = None,
241        vmin: Optional[float] = None,
242        vmax: Optional[float] = None,
243        cmap: Optional[str] = None,
244        camera_position: Optional[str] = None,
245        title: Optional[str] = None,
246        component: Union[str, int] = "magnitude",
247    ) -> None:
248        """
249        Open an interactive, rotatable 3D PyVista window for a single e-field.
250
251        The window is fully interactive (rotate, zoom, pan) and blocks until
252        closed.  No extra dependencies beyond PyVista are required.
253
254        Parameters
255        ----------
256        efield_path :
257            Path to the e-field NIfTI file.
258        mask_path :
259            Optional binary mask to colocalize (e.g.
260            ``m2m_<sub>/surfaces/cereb_mask.nii.gz``).  Rendered as a
261            semi-transparent coloured surface.
262        mask_color :
263            Colour of the mask surface (default ``'cyan'``).
264        mask_opacity :
265            Opacity of the mask surface in [0, 1] (default 0.3).
266        threshold_percentile :
267            Percentile cutoff for non-zero voxels.  Defaults to
268            ``self.threshold_percentile``.
269        vmin, vmax :
270            Explicit colour scale limits.  Auto-scales if both ``None``.
271        cmap :
272            Colormap override.  Defaults to ``self.cmap``.
273        camera_position :
274            Initial camera position override.  Defaults to
275            ``self.camera_position``.
276        title :
277            Window title.  Defaults to the e-field filename stem.
278
279        Examples
280        --------
281        >>> viz = Visualizer(output_dir="output/")
282        >>> viz.view_efield_interactive(
283        ...     efield_path="sub-0011_magnE.nii.gz",
284        ...     mask_path="m2m_0011/surfaces/cereb_mask.nii.gz",
285        ...     mask_color="cyan",
286        ...     mask_opacity=0.3,
287        ... )
288        """
289        plotter = self._build_plotter(
290            efield_path=Path(efield_path),
291            camera_position=camera_position or self.camera_position,
292            cmap=cmap or self.cmap,
293            threshold_percentile=(
294                threshold_percentile
295                if threshold_percentile is not None
296                else self.threshold_percentile
297            ),
298            vmin=vmin,
299            vmax=vmax,
300            mask_path=Path(mask_path) if mask_path is not None else None,
301            mask_color=mask_color,
302            mask_opacity=mask_opacity,
303            off_screen=False,
304            component=component,
305        )
306        comp_label = {0: " [Ex]", 1: " [Ey]", 2: " [Ez]"}.get(component, "")
307        plotter.title = (title or Path(efield_path).stem) + comp_label
308        logger.info(f"Opening interactive 3D viewer: {efield_path}")
309        plotter.show()

311    def efields_figures(
312        self,
313        file_info_by_roi_mode: Dict[Tuple[str, str], List[Tuple[str, Path]]],
314        t1_brain_by_subject: Optional[Dict[str, Path]] = None,
315        space: str = "mni",
316    ) -> None:
317        """
318        Generate one 3D figure per (roi, mode) pair.
319
320        Parameters
321        ----------
322        file_info_by_roi_mode :
323            Mapping ``(roi, mode) → [(subject, efield_path), ...]``.
324        t1_brain_by_subject :
325            Optional mapping ``subject → T1 brain path``. Brain surface is
326            rendered behind the e-field.  Must be in the same space as the
327            e-fields (use ``T1_MNI_brain.nii.gz`` for MNI,
328            ``T1_subject_brain.nii.gz`` for subject space).
329        space : str
330            ``'mni'`` or ``'native'`` — included in the output filename so
331            figures from both spaces are saved without overwriting each other.
332        """
333        output_dir = self.output_dir / "1-simu"
334        output_dir.mkdir(parents=True, exist_ok=True)
335        tag = space_tag(space)
336
337        # Global colour scale across all files
338        all_nonzero: List[np.ndarray] = []
339        for subject_files in file_info_by_roi_mode.values():
340            for _, efield_path in subject_files:
341                data = np.squeeze(nib.load(str(efield_path)).get_fdata())
342                nonzero = data[data > 0]
343                if nonzero.size > 0:
344                    if self.threshold_percentile > 0:
345                        thresh = np.percentile(nonzero, self.threshold_percentile)
346                        nonzero = nonzero[nonzero >= thresh]
347                    all_nonzero.append(nonzero)
348
349        if not all_nonzero:
350            logger.warning("No non-zero data found, skipping efields_figures.")
351            return
352
353        all_values = np.concatenate(all_nonzero)
354        vmin, vmax = float(np.min(all_values)), float(np.max(all_values))
355        logger.info(f"Global colour scale: {vmin:.3f} – {vmax:.3f} V/m")
356
357        for (roi, mode), subject_files in file_info_by_roi_mode.items():
358            logger.info(f"Generating figure: {roi} – {mode}")
359            n_subjects = len(subject_files)
360            n_cols = min(6, n_subjects)
361            n_rows = (n_subjects + n_cols - 1) // n_cols
362
363            fig, axes = plt.subplots(n_rows, n_cols, figsize=(4 * n_cols, 4 * n_rows))
364            if n_subjects == 1:
365                axes = np.array([[axes]])
366            elif n_rows == 1:
367                axes = axes.reshape(1, -1)
368            elif n_cols == 1:
369                axes = axes.reshape(-1, 1)
370
371            for idx, (subject, efield_path) in enumerate(subject_files):
372                row, col = divmod(idx, n_cols)
373                ax = axes[row, col]
374                image = self._create_3d_view(
375                    efield_path=efield_path,
376                    camera_position=self.camera_position,
377                    cmap=self.cmap,
378                    threshold_percentile=self.threshold_percentile,
379                    vmin=vmin,
380                    vmax=vmax,
381                )
382                ax.imshow(image)
383                ax.axis("off")
384                ax.set_title(subject, fontsize=12)
385
386            for idx in range(n_subjects, n_rows * n_cols):
387                row, col = divmod(idx, n_cols)
388                axes[row, col].axis("off")
389
390            fig.suptitle(
391                f"{roi} – {mode} ({space.upper()})", fontsize=16, fontweight="bold"
392            )
393            plt.tight_layout()
394            out_path = (
395                output_dir / f"efields_3d_{roi}_{mode}_{tag}_{self.camera_position}.png"
396            )
397            save_figure(
398                out_path, if_exists=self.if_exists, dpi=300, bbox_inches="tight"
399            )
400            logger.info(f"  Saved: {out_path}")
401
402        logger.info(f"All 3D figures saved in {output_dir}")

404    def efields_histograms(
405        self,
406        data_by_subject: Dict[str, List[Tuple[str, str, Path, Path]]],
407        region: str = "intra",
408        space: str = "mni",
409    ) -> None:
410        """
411        Generate one histogram figure per subject comparing masked vs cleaned e-fields.
412
413        Parameters
414        ----------
415        data_by_subject :
416            Mapping ``subject → [(roi, mode, masked_path, cleaned_path), ...]``.
417        region :
418            Label included in the figure title and filename (``"intra"`` or ``"extra"``).
419        space :
420            ``'mni'`` or ``'native'`` — filename suffix.
421        """
422        output_dir = self.output_dir / "2-preprocess"
423        output_dir.mkdir(parents=True, exist_ok=True)
424        tag = space_tag(space)
425
426        for subject, subject_data in data_by_subject.items():
427            logger.info(f"{region}-ROI histograms for {subject}")
428            n_plots = len(subject_data)
429            n_cols = min(3, n_plots)
430            n_rows = (n_plots + n_cols - 1) // n_cols
431
432            fig, axes = plt.subplots(n_rows, n_cols, figsize=(5 * n_cols, 4 * n_rows))
433            if n_plots == 1:
434                axes = np.array([[axes]])
435            elif n_rows == 1:
436                axes = axes.reshape(1, -1)
437            elif n_cols == 1:
438                axes = axes.reshape(-1, 1)
439
440            for idx, (roi, mode, masked_path, cleaned_path) in enumerate(subject_data):
441                row_idx, col = divmod(idx, n_cols)
442
443                masked_data = nib.load(str(masked_path)).get_fdata().ravel()
444                masked_data = masked_data[masked_data > 0]
445                cleaned_data = nib.load(str(cleaned_path)).get_fdata().ravel()
446                cleaned_data = cleaned_data[
447                    np.isfinite(cleaned_data) & (cleaned_data > 0)
448                ]
449
450                ax = axes[row_idx, col]
451                if masked_data.size > 0 or cleaned_data.size > 0:
452                    all_vals = np.concatenate(
453                        [a for a in [masked_data, cleaned_data] if a.size > 0]
454                    )
455                    xrange = (float(all_vals.min()), float(all_vals.max()))
456                else:
457                    xrange = None
458                if masked_data.size > 0:
459                    ax.hist(
460                        masked_data,
461                        bins=self.bins,
462                        range=xrange,
463                        alpha=0.6,
464                        label="Before",
465                        color="red",
466                        density=True,
467                    )
468                if cleaned_data.size > 0:
469                    ax.hist(
470                        cleaned_data,
471                        bins=self.bins,
472                        range=xrange,
473                        alpha=0.6,
474                        label="After",
475                        color="blue",
476                        density=True,
477                    )
478                if masked_data.size == 0 and cleaned_data.size == 0:
479                    ax.text(
480                        0.5,
481                        0.5,
482                        "No data",
483                        ha="center",
484                        va="center",
485                        transform=ax.transAxes,
486                    )
487                ax.set_xlabel("E-field (V/m)", fontsize=10)
488                ax.set_ylabel("Density", fontsize=10)
489                ax.set_title(f"{roi} | {mode}", fontsize=11)
490                ax.legend(fontsize=8)
491                ax.grid(True, alpha=0.3)
492
493            for idx in range(n_plots, n_rows * n_cols):
494                row_idx, col = divmod(idx, n_cols)
495                axes[row_idx, col].axis("off")
496
497            fig.suptitle(
498                f"Preprocessing {region}-ROI – {subject}",
499                fontsize=16,
500                fontweight="bold",
501            )
502            plt.tight_layout()
503            out_path = output_dir / f"efields_histograms_{subject}_{region}_{tag}.png"
504            save_figure(
505                out_path, if_exists=self.if_exists, dpi=300, bbox_inches="tight"
506            )
507            logger.info(f"  Saved: {out_path}")
508
509        logger.info(f"All histograms saved in {output_dir}")

106            ``'magnitude'`` (default) → computes ‖E‖ = √(Ex²+Ey²+Ez²).

136                    f"{label} (vol[{component}]), signed"
137                )

176            mask_origin = mask_img.affine[:3, 3]

178            mask_grid.dimensions = np.array(mask_data.shape) + 1

180            mask_grid.origin = mask_origin
181            mask_grid.cell_data["mask"] = mask_data.flatten(order="F")
