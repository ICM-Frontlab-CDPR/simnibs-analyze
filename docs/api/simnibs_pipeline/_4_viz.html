 20    Generates all pipeline visualisations.
 21
 22    Rendering parameters (colormap, thresholds, etc.) are set once at
 23    construction time and reused across all methods.
 24

 28        Base output directory.  Each method writes into a named sub-directory

 31        Matplotlib / PyVista colormap for e-field figures.
 32    threshold_percentile :
 33        Non-zero voxel percentile below which values are zeroed in 3D renders.
 34    bins :
 35        Number of histogram bins for preprocessing histograms.
 36    camera_position :
 37        PyVista camera position string (``'xy'``, ``'xz'``, ``'yz'``).

 58    @staticmethod
 59    def _create_3d_view(
 60        efield_path: Path,
 61        camera_position: str = "xy",
 62        cmap: str = "hot",
 63        threshold_percentile: float = 0.0,
 64        vmin: Optional[float] = None,
 65        vmax: Optional[float] = None,
 66        brain_bg_path: Optional[Path] = None,
 67    ) -> np.ndarray:
 68        """
 69        Render an e-field NIfTI volume with PyVista (offscreen) and return an RGBA array.
 70
 71        When ``brain_bg_path`` is provided, the brain surface is rendered as a
 72        semi-transparent white mesh and the e-field is overlaid as a coloured
 73        volume.  **Both must be in the same coordinate space** — for MNI e-fields
 74        (``*_scalar_MNI_magnE.nii.gz``), pass the ``T1_MNI_brain.nii.gz`` produced
 75        by :meth:`AnatomicalPreparer._make_mni_brain_bg`.  No resampling is
 76        performed here.
 77
 78        Parameters
 79        ----------
 80        efield_path :
 81            Path to the e-field NIfTI file.
 82        camera_position :
 83            PyVista camera position string (``'xy'``, ``'xz'``, ``'yz'``).
 84        cmap :
 85            Colormap name.
 86        threshold_percentile :
 87            Voxels below this percentile of non-zero values are zeroed.
 88        vmin, vmax :
 89            Explicit colour scale limits. If both ``None``, PyVista auto-scales.
 90        brain_bg_path :
 91            Optional path to a skull-stripped T1 **in the same space as the
 92            e-field**.  Rendered as a semi-transparent brain surface behind the
 93            e-field volume.
 94        """
 95        efield_img = nib.as_closest_canonical(nib.load(str(efield_path)))
 96        data = np.squeeze(efield_img.get_fdata())
 97        if threshold_percentile > 0:
 98            nonzero = data[data > 0]
 99            if nonzero.size > 0:
100                thresh = np.percentile(nonzero, threshold_percentile)
101                data[data < thresh] = 0
102
103        efield_affine = efield_img.affine
104        efield_spacing = efield_img.header.get_zooms()[:3]
105        efield_origin = efield_affine[:3, 3]
106
107        grid = pv.ImageData()
108        grid.dimensions = np.array(data.shape) + 1
109        grid.spacing = efield_spacing
110        grid.origin = efield_origin
111        grid.cell_data["values"] = data.flatten(order="F")
112
113        plotter = pv.Plotter(off_screen=True)
114
115        # ── Fond anatomique (surface du cerveau) ──────────────────────────
116        if brain_bg_path is not None:
117            t1_img = nib.as_closest_canonical(nib.load(str(brain_bg_path)))
118            t1_data = np.squeeze(t1_img.get_fdata())
119            t1_spacing = t1_img.header.get_zooms()[:3]
120            t1_origin = t1_img.affine[:3, 3]
121            t1_grid = pv.ImageData()
122            t1_grid.dimensions = np.array(t1_data.shape) + 1
123            t1_grid.spacing = t1_spacing
124            t1_grid.origin = t1_origin
125            t1_grid.cell_data["t1"] = t1_data.flatten(order="F")
126            nonzero_t1 = t1_data[t1_data > 0]
127            iso_val = float(np.percentile(nonzero_t1, 20)) if nonzero_t1.size > 0 else 0.1
128            brain_surface = t1_grid.cell_data_to_point_data().contour([iso_val])
129            plotter.add_mesh(brain_surface, color="white", opacity=0.15, smooth_shading=True)
130
131        # ── Volume e-field ────────────────────────────────────────────────
132        if vmin is not None and vmax is not None:
133            plotter.add_volume(grid, cmap=cmap, clim=[vmin, vmax])
134        else:
135            plotter.add_volume(grid, cmap=cmap)
136
137        plotter.camera_position = camera_position
138        image = plotter.screenshot(return_img=True)
139        plotter.close()
140        return image

145    def efields_figures(
146        self,
147        file_info_by_roi_mode: Dict[Tuple[str, str], List[Tuple[str, Path]]],
148        t1_brain_by_subject: Optional[Dict[str, Path]] = None,
149        space: str = "mni",
150    ) -> None:
151        """
152        Generate one 3D figure per (roi, mode) pair.
153
154        Parameters
155        ----------
156        file_info_by_roi_mode :
157            Mapping ``(roi, mode) → [(subject, efield_path), ...]``.
158        t1_brain_by_subject :
159            Optional mapping ``subject → T1 brain path``. Brain surface is
160            rendered behind the e-field.  Must be in the same space as the
161            e-fields (use ``T1_MNI_brain.nii.gz`` for MNI,
162            ``T1_subject_brain.nii.gz`` for subject space).
163        space : str
164            ``'mni'`` or ``'subject'`` — included in the output filename so
165            figures from both spaces are saved without overwriting each other.
166        """
167        output_dir = self.output_dir / "simu"
168        output_dir.mkdir(parents=True, exist_ok=True)
169
170        # Global colour scale across all files
171        all_nonzero: List[np.ndarray] = []
172        for subject_files in file_info_by_roi_mode.values():
173            for _, efield_path in subject_files:
174                data = np.squeeze(nib.load(str(efield_path)).get_fdata())
175                nonzero = data[data > 0]
176                if nonzero.size > 0:
177                    if self.threshold_percentile > 0:
178                        thresh = np.percentile(nonzero, self.threshold_percentile)
179                        nonzero = nonzero[nonzero >= thresh]
180                    all_nonzero.append(nonzero)
181
182        if not all_nonzero:
183            logger.warning("Aucune donnée non-nulle trouvée, abandon de efields_figures.")
184            return
185
186        all_values = np.concatenate(all_nonzero)
187        vmin, vmax = float(np.min(all_values)), float(np.max(all_values))
188        logger.info(f"Échelle de couleur globale : {vmin:.3f} – {vmax:.3f} V/m")
189
190        for (roi, mode), subject_files in file_info_by_roi_mode.items():
191            logger.info(f"Génération de la figure : {roi} – {mode}")
192            n_subjects = len(subject_files)
193            n_cols = min(6, n_subjects)
194            n_rows = (n_subjects + n_cols - 1) // n_cols
195
196            fig, axes = plt.subplots(n_rows, n_cols, figsize=(4 * n_cols, 4 * n_rows))
197            if n_subjects == 1:
198                axes = np.array([[axes]])
199            elif n_rows == 1:
200                axes = axes.reshape(1, -1)
201            elif n_cols == 1:
202                axes = axes.reshape(-1, 1)
203
204            for idx, (subject, efield_path) in enumerate(subject_files):
205                row, col = divmod(idx, n_cols)
206                ax = axes[row, col]
207                brain_bg = (t1_brain_by_subject or {}).get(subject)
208                image = self._create_3d_view(
209                    efield_path=efield_path,
210                    camera_position=self.camera_position,
211                    cmap=self.cmap,
212                    threshold_percentile=self.threshold_percentile,
213                    vmin=vmin,
214                    vmax=vmax,
215                    brain_bg_path=brain_bg,
216                )
217                ax.imshow(image)
218                ax.axis("off")
219                ax.set_title(subject, fontsize=12)
220
221            for idx in range(n_subjects, n_rows * n_cols):
222                row, col = divmod(idx, n_cols)
223                axes[row, col].axis("off")
224
225            fig.suptitle(f"{roi} – {mode} ({space.upper()})", fontsize=16, fontweight="bold")
226            plt.tight_layout()
227            out_path = output_dir / f"efields_3d_{roi}_{mode}_{space}_{self.camera_position}.png"
228            plt.savefig(out_path, dpi=300, bbox_inches="tight")
229            plt.close()
230            logger.info(f"  Sauvegardé : {out_path}")
231
232        logger.info(f"Toutes les figures 3D dans {output_dir}")

234    def efields_histograms(
235        self,
236        data_by_subject: Dict[str, List[Tuple[str, str, Path, Path]]],
237        region: str = "intra",
238    ) -> None:
239        """
240        Generate one histogram figure per subject comparing masked vs cleaned e-fields.
241
242        Parameters
243        ----------
244        data_by_subject :
245            Mapping ``subject → [(roi, mode, masked_path, cleaned_path), ...]``.
246        region :
247            Label inclus dans le titre et le nom de fichier (``"intra"`` ou ``"extra"``).
248        """
249        output_dir = self.output_dir / "preprocess"
250        output_dir.mkdir(parents=True, exist_ok=True)
251
252        for subject, subject_data in data_by_subject.items():
253            logger.info(f"Histogrammes {region}-ROI pour {subject}")
254            n_plots = len(subject_data)
255            n_cols = min(3, n_plots)
256            n_rows = (n_plots + n_cols - 1) // n_cols
257
258            fig, axes = plt.subplots(n_rows, n_cols, figsize=(5 * n_cols, 4 * n_rows))
259            if n_plots == 1:
260                axes = np.array([[axes]])
261            elif n_rows == 1:
262                axes = axes.reshape(1, -1)
263            elif n_cols == 1:
264                axes = axes.reshape(-1, 1)
265
266            for idx, (roi, mode, masked_path, cleaned_path) in enumerate(subject_data):
267                row_idx, col = divmod(idx, n_cols)
268
269                masked_data = nib.load(str(masked_path)).get_fdata().ravel()
270                masked_data = masked_data[masked_data > 0]
271                cleaned_data = nib.load(str(cleaned_path)).get_fdata().ravel()
272                cleaned_data = cleaned_data[np.isfinite(cleaned_data) & (cleaned_data > 0)]
273
274                ax = axes[row_idx, col]
275                if masked_data.size > 0 or cleaned_data.size > 0:
276                    all_vals = np.concatenate([a for a in [masked_data, cleaned_data] if a.size > 0])
277                    xrange = (float(all_vals.min()), float(all_vals.max()))
278                else:
279                    xrange = None
280                if masked_data.size > 0:
281                    ax.hist(masked_data, bins=self.bins, range=xrange, alpha=0.6, label="Avant", color="red", density=True)
282                if cleaned_data.size > 0:
283                    ax.hist(cleaned_data, bins=self.bins, range=xrange, alpha=0.6, label="Après", color="blue", density=True)
284                if masked_data.size == 0 and cleaned_data.size == 0:
285                    ax.text(0.5, 0.5, "Aucune donnée", ha="center", va="center", transform=ax.transAxes)
286                ax.set_xlabel("E-field (V/m)", fontsize=10)
287                ax.set_ylabel("Densité", fontsize=10)
288                ax.set_title(f"{roi} | {mode}", fontsize=11)
289                ax.legend(fontsize=8)
290                ax.grid(True, alpha=0.3)
291
292            for idx in range(n_plots, n_rows * n_cols):
293                row_idx, col = divmod(idx, n_cols)
294                axes[row_idx, col].axis("off")
295
296            fig.suptitle(
297                f"Preprocessing {region}-ROI – {subject}",
298                fontsize=16,
299                fontweight="bold",
300            )
301            plt.tight_layout()
302            out_path = output_dir / f"efields_histograms_{subject}_{region}.png"
303            plt.savefig(out_path, dpi=300, bbox_inches="tight")
304            plt.close()
305            logger.info(f"  Sauvegardé : {out_path}")
306
307        logger.info(f"Tous les histogrammes dans {output_dir}")

388    def plot_simulation_vs_optimization(
389        self,
390        df: pd.DataFrame,
391        metric: str = "mean",
392        subject_col: str = "subject",
393        condition_col: str = "condition",
394    ) -> None:
395        """
396        Scatter plot: per subject, simulation (x) vs optimisation (y) for each ROI.
397
398        Parameters
399        ----------
400        df :
401            Features DataFrame whose ``condition`` column encodes
402            ``<roi>_simulation`` / ``<roi>_optimization`` labels.
403        metric :
404            Column used for both axes.
405        subject_col :
406            Column containing subject identifiers.
407        condition_col :
408            Column containing condition labels.
409        """
410        output_dir = self.output_dir / "figures"
411        output_dir.mkdir(parents=True, exist_ok=True)
412
413        df = df.copy()
414        df["type"] = df[condition_col].apply(
415            lambda x: "optimization" if "optimization" in x else "simulation"
416        )
417        df["roi"] = df[condition_col].apply(
418            lambda x: x.replace("_simulation", "").replace("_optimization", "")
419        )
420
421        rois = df["roi"].unique()
422        n_rois = len(rois)
423
424        fig, axes = plt.subplots(1, n_rois, figsize=(6 * n_rois, 5))
425        if n_rois == 1:
426            axes = [axes]
427
428        for i, roi in enumerate(rois):
429            df_roi = df[df["roi"] == roi]
430            pivot = df_roi.pivot_table(
431                index=subject_col, columns="type", values=metric, aggfunc="first"
432            ).dropna()
433
434            if pivot.empty or "simulation" not in pivot.columns or "optimization" not in pivot.columns:
435                continue
436
437            ax = axes[i]
438            ax.scatter(pivot["simulation"], pivot["optimization"], alpha=0.7, s=100)
439
440            for subject, row in pivot.iterrows():
441                ax.annotate(
442                    subject,
443                    (row["simulation"], row["optimization"]),
444                    fontsize=8,
445                    alpha=0.6,
446                    xytext=(3, 3),
447                    textcoords="offset points",
448                )
449
450            min_val = min(pivot["simulation"].min(), pivot["optimization"].min())
451            max_val = max(pivot["simulation"].max(), pivot["optimization"].max())
452            ax.plot([min_val, max_val], [min_val, max_val], "k--", alpha=0.3, label="y=x")
453            ax.set_xlabel(f"Simulation {metric} (V/m)")
454            ax.set_ylabel(f"Optimization {metric} (V/m)")
455            ax.set_title(f"ROI: {roi}")
456            ax.legend()
457            ax.grid(True, alpha=0.3)
458            ax.set_aspect("equal", adjustable="box")
459
460        plt.tight_layout()
461        out_path = output_dir / "simulation_vs_optimization.png"
462        plt.savefig(out_path, dpi=150, bbox_inches="tight")
463        plt.close()
464        logger.info(f"Sauvegardé : {out_path}")

128            iso_val = float(np.percentile(nonzero_t1, 20)) if nonzero_t1.size > 0 else 0.1
129            brain_surface = t1_grid.cell_data_to_point_data().contour([iso_val])
130            plotter.add_mesh(brain_surface, color="white", opacity=0.15, smooth_shading=True)
131
132        # ── Volume e-field ────────────────────────────────────────────────
133        if vmin is not None and vmax is not None:
134            plotter.add_volume(grid, cmap=cmap, clim=[vmin, vmax])
135        else:
136            plotter.add_volume(grid, cmap=cmap)
137
138        plotter.camera_position = camera_position
139        image = plotter.screenshot(return_img=True)
140        plotter.close()

390        self,
391        df: pd.DataFrame,
392        metric: str = "mean",
393        subject_col: str = "subject",
394        condition_col: str = "condition",
395    ) -> None:
396        """
397        Scatter plot: per subject, simulation (x) vs optimisation (y) for each ROI.
398
399        Parameters
400        ----------
401        df :
402            Features DataFrame whose ``condition`` column encodes
403            ``<roi>_simulation`` / ``<roi>_optimization`` labels.
404        metric :
405            Column used for both axes.
406        subject_col :
407            Column containing subject identifiers.
408        condition_col :
409            Column containing condition labels.
410        """
411        output_dir = self.output_dir / "figures"
412        output_dir.mkdir(parents=True, exist_ok=True)
413
414        df = df.copy()
415        df["type"] = df[condition_col].apply(
416            lambda x: "optimization" if "optimization" in x else "simulation"
417        )
418        df["roi"] = df[condition_col].apply(
419            lambda x: x.replace("_simulation", "").replace("_optimization", "")
420        )
421
422        rois = df["roi"].unique()
423        n_rois = len(rois)
424

429        for i, roi in enumerate(rois):
430            df_roi = df[df["roi"] == roi]
431            pivot = df_roi.pivot_table(
432                index=subject_col, columns="type", values=metric, aggfunc="first"
433            ).dropna()
434
435            if pivot.empty or "simulation" not in pivot.columns or "optimization" not in pivot.columns:
436                continue
437
438            ax = axes[i]
439            ax.scatter(pivot["simulation"], pivot["optimization"], alpha=0.7, s=100)

441            for subject, row in pivot.iterrows():

449                )

451            min_val = min(pivot["simulation"].min(), pivot["optimization"].min())
452            max_val = max(pivot["simulation"].max(), pivot["optimization"].max())
453            ax.plot([min_val, max_val], [min_val, max_val], "k--", alpha=0.3, label="y=x")
454            ax.set_xlabel(f"Simulation {metric} (V/m)")
